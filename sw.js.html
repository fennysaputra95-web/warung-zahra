// Nama cache aplikasi const CACHE_NAME = 'warung-zahra-v1'; // Daftar file & resource yang harus disimpan secara offline const ASSETS_TO_CACHE = [ './', './index.html', // Sesuaikan jika nama file HTML Anda berbeda 'https://cdn.tailwindcss.com', 'https://unpkg.com/html5-qrcode' ]; // 1. Proses Install: Menyimpan file ke cache browser self.addEventListener('install', (event) => { event.waitUntil( caches.open(CACHE_NAME).then((cache) => { console.log('[Service Worker] Caching all assets'); return cache.addAll(ASSETS_TO_CACHE); }) ); self.skipWaiting(); }); // 2. Proses Aktivasi: Membersihkan cache lama jika ada update self.addEventListener('activate', (event) => { event.waitUntil( caches.keys().then((cacheNames) => { return Promise.all( cacheNames.map((cache) => { if (cache !== CACHE_NAME) { console.log('[Service Worker] Deleting old cache:', cache); return caches.delete(cache); } }) ); }) ); self.clients.claim(); }); // 3. Proses Fetch: Ambil data dari cache jika sedang offline self.addEventListener('fetch', (event) => { event.respondWith( caches.match(event.request).then((cachedResponse) => { // Jika ada di cache, gunakan file offline. Jika tidak, ambil dari jaringan return cachedResponse || fetch(event.request); }) ); });
